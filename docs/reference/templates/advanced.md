# Advanced Template Customization

This guide covers advanced techniques for customizing the BenchPRO template system.

## Custom Block Creation

### Creating Custom String Blocks

```python
from benchpro.templates.blocks import StringTemplateBlock
from benchpro.templates.standard_blocks import PRIORITY_HEADER_COMMENTS

custom_header = StringTemplateBlock(
    name="custom_header",
    content="# Generated by {{ user }} on {{ date }}",
    priority=PRIORITY_HEADER_COMMENTS,
    description="Custom header with user and date",
)
```

### Creating Custom Function Blocks

Function blocks allow you to generate content programmatically:

```python
from benchpro.templates.blocks import FunctionTemplateBlock
from benchpro.templates.standard_blocks import PRIORITY_MODULE_LOADING

def generate_modules(context, execution_context):
    """Dynamically generate module load commands."""
    if "modules" not in context:
        return ""
    
    lines = ["# Dynamically generated module load commands"]
    for module in context["modules"]:
        version = context.get("module_versions", {}).get(module, "")
        if version:
            lines.append(f"module load {module}/{version}")
        else:
            lines.append(f"module load {module}")
    
    return "\n".join(lines)

module_block = FunctionTemplateBlock(
    name="dynamic_modules",
    function=generate_modules,
    priority=PRIORITY_MODULE_LOADING,
    description="Dynamically generated module load commands"
)
```

## Custom Script Generators

You can create custom script generators for specialized execution contexts:

```python
from benchpro.templates.script_generators import ComposableScriptGenerator
from benchpro.templates.standard_blocks import register_standard_blocks

class GPUScriptGenerator(ComposableScriptGenerator):
    """Script generator for GPU-enabled execution."""
    
    def __init__(self):
        super().__init__(execution_context="gpu")
        
        # Add GPU-specific blocks
        gpu_setup_block = StringTemplateBlock(
            name="gpu_setup",
            content="""# GPU Setup
export CUDA_VISIBLE_DEVICES={{ gpu.devices if gpu.devices else "0" }}
export GPU_MEMORY_FRACTION={{ gpu.memory_fraction if gpu.memory_fraction else "1.0" }}
""",
            priority=55,  # After environment variables, before user commands
            description="GPU environment setup"
        )
        
        self.composition_engine.register_block(gpu_setup_block)
    
    def prepare_variables(self, config):
        """Add GPU-specific variables to the template context."""
        variables = super().prepare_variables(config)
        
        # Set default GPU config if not present
        if "gpu" not in variables:
            variables["gpu"] = {
                "devices": "0",
                "memory_fraction": "1.0"
            }
        
        return variables
```

## Extending the Composition Engine

For more advanced customization, you can extend the composition engine:

```python
from benchpro.templates.composition import TemplateCompositionEngine

class ExtendedCompositionEngine(TemplateCompositionEngine):
    """Extended template composition engine with additional features."""
    
    def __init__(self):
        super().__init__()
        self.block_groups = {}  # name -> [blocks]
    
    def register_block_group(self, group_name, blocks):
        """Register a group of blocks together."""
        self.block_groups[group_name] = blocks
        for block in blocks:
            self.register_block(block)
    
    def unregister_block_group(self, group_name):
        """Unregister a group of blocks."""
        if group_name in self.block_groups:
            for block in self.block_groups[group_name]:
                self.unregister_block(block.name)
            del self.block_groups[group_name]
    
    def compose_with_hooks(self, context, execution_context, pre_hooks=None, post_hooks=None):
        """Compose with pre and post processing hooks."""
        pre_hooks = pre_hooks or []
        post_hooks = post_hooks or []
        
        # Apply pre-processing hooks
        for hook in pre_hooks:
            context = hook(context, execution_context)
        
        # Compose the script
        script = self.compose(context, execution_context)
        
        # Apply post-processing hooks
        for hook in post_hooks:
            script = hook(script, context, execution_context)
        
        return script
```

## Integration with External Systems

The template system can be integrated with external systems:

```python
import requests
from benchpro.templates.blocks import FunctionTemplateBlock

def fetch_remote_template(context, execution_context):
    """Fetch a template from a remote source."""
    if "remote_template_url" not in context:
        return ""
    
    try:
        response = requests.get(context["remote_template_url"])
        response.raise_for_status()
        return response.text
    except Exception as e:
        return f"# Error fetching remote template: {str(e)}"

remote_block = FunctionTemplateBlock(
    name="remote_template",
    function=fetch_remote_template,
    priority=90,  # Just before user commands
    description="Fetches a template from a remote source"
)
```
